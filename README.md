    function renderAllProducts(filterSlug = 'all', searchText = '') {
      const normalizedSearch = searchText.trim().toLowerCase();

      const filtered = PRODUCTS_DATA.filter(item => {
        const matchCat = filterSlug === 'all' || item.categorySlug === filterSlug;
        const matchQuery = !normalizedSearch ||
          item.title.toLowerCase().includes(normalizedSearch) ||
          item.categoryName.toLowerCase().includes(normalizedSearch);
        return matchCat && matchQuery;
      });

      const buddhaItems = filtered.filter(p => p.section === 'buddha');
      const mascotItems = filtered.filter(p => p.section === 'mascot');
      const archItems = filtered.filter(p => p.section === 'architecture');

      const secBuddha = document.getElementById('san-pham-tuong-phat');
      const secMascot = document.getElementById('san-pham-cong-giao-linh-vat');
      const secArch = document.getElementById('san-pham-do-tho-kien-truc');

      document.getElementById('productGridBuddha').innerHTML = buddhaItems.map(createFlatsomeProductCard).join('');
      document.getElementById('productGridMascot').innerHTML = mascotItems.map(createFlatsomeProductCard).join('');
      document.getElementById('productGridArchitecture').innerHTML = archItems.map(createFlatsomeProductCard).join('');

      secBuddha.style.display = buddhaItems.length ? 'block' : 'none';
      secMascot.style.display = mascotItems.length ? 'block' : 'none';
      secArch.style.display = archItems.length ? 'block' : 'none';
    }

    function handleCategoryFilter(slug) {
      const banner = document.getElementById('filterStatusBanner');
      const textEl = document.getElementById('filterStatusText');
      const selectEl = document.getElementById('searchCategorySelect');
      if (selectEl) selectEl.value = slug;

      renderAllProducts(slug, document.getElementById('headerSearchInput')?.value || '');

      if (slug === 'all') {
        banner.classList.add('hidden');
      } else {
        const foundCat = CATEGORIES_DATA.find(c => c.slug === slug);
        textEl.textContent = `Đang hiển thị danh mục: ${foundCat ? foundCat.name : slug}`;
        banner.classList.remove('hidden');
        banner.scrollIntoView({ behavior: 'smooth', block: 'center' });
      }
    }

    function handleSearchProducts(query) {
      const currentCat = document.getElementById('searchCategorySelect')?.value || 'all';
      renderAllProducts(currentCat, query);
    }

    // 5. SLIDER 2 ẢNH BANNER TỰ TRƯỢT (AUTO SLIDE 4.5 GIÂY)
    // ---------------------------------------------------------------------
    let currentBannerIndex = 0;
    const totalBannerSlides = 2;
    let bannerInterval = null;

    function updateBannerUI() {
      for (let i = 0; i < totalBannerSlides; i++) {
        const slide = document.getElementById(`banner-slide-${i}`);
        const dot = document.getElementById(`banner-dot-${i}`);
        if (i === currentBannerIndex) {
          slide.classList.remove('opacity-0', 'z-0');
          slide.classList.add('opacity-100', 'z-10');
          dot.className = 'w-8 h-2.5 rounded-full bg-brand-gold transition-all';
        } else {
          slide.classList.remove('opacity-100', 'z-10');
          slide.classList.add('opacity-0', 'z-0');
          dot.className = 'w-2.5 h-2.5 rounded-full bg-white/60 transition-all';
        }
      }
    }

    function changeBannerSlide(step) {
      currentBannerIndex = (currentBannerIndex + step + totalBannerSlides) % totalBannerSlides;
      updateBannerUI();
      resetBannerTimer();
    }

    function goToBannerSlide(index) {
      currentBannerIndex = index;
      updateBannerUI();
      resetBannerTimer();
    }

    function resetBannerTimer() {
      if (bannerInterval) clearInterval(bannerInterval);
      bannerInterval = setInterval(() => {
        currentBannerIndex = (currentBannerIndex + 1) % totalBannerSlides;
        updateBannerUI();
      }, 4500);
    }
